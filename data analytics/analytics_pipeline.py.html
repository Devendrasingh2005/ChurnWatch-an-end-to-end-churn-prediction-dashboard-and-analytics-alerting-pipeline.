"""Web analytics ETL: extract -> validate -> load -> alert.

Run daily:  0 6 * * *  python analytics_pipeline.py
Env: SOURCE=mock|ga4  GA4_PROPERTY, GA4_TOKEN  WAREHOUSE=sqlite|bigquery
     SQLITE_PATH, BQ_TABLE (project.dataset.table)  SLACK_WEBHOOK
"""
import os, time, logging, sqlite3, datetime as dt
import numpy as np, pandas as pd, requests

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
log = logging.getLogger("pipeline")
DAYS, DROP_THRESHOLD, Z_LIMIT = 30, 0.30, 3.0

def retry(fn, tries=3, wait=2):
    for i in range(tries):
        try: return fn()
        except Exception as e:
            log.warning("attempt %d/%d failed: %s", i + 1, tries, e)
            if i == tries - 1: raise
            time.sleep(wait * 2 ** i)

# ---------- extract ----------
def extract_ga4():
    url = f"https://analyticsdata.googleapis.com/v1beta/properties/{os.environ['GA4_PROPERTY']}:runReport"
    body = {"dateRanges": [{"startDate": f"{DAYS}daysAgo", "endDate": "yesterday"}],
            "dimensions": [{"name": "date"}, {"name": "sessionDefaultChannelGroup"}],
            "metrics": [{"name": m} for m in ("sessions", "activeUsers", "conversions", "bounceRate")]}
    r = requests.post(url, json=body, timeout=30,
                      headers={"Authorization": f"Bearer {os.environ['GA4_TOKEN']}"})
    r.raise_for_status()
    rows = [[d["value"] for d in x["dimensionValues"] + x["metricValues"]] for x in r.json().get("rows", [])]
    return pd.DataFrame(rows, columns=["date", "channel", "sessions", "users", "conversions", "bounce_rate"])

def extract_mock():
    rng = np.random.default_rng(int(time.time()) // 86400)
    days = pd.date_range(end=dt.date.today() - dt.timedelta(days=1), periods=DAYS)
    rows = []
    for d in days:
        for ch, base in {"Organic": 900, "Paid": 500, "Direct": 400, "Email": 150}.items():
            s = int(rng.normal(base, base * .08))
            if d == days[-1] and ch == "Organic": s = int(s * .4)  # simulated outage
            rows.append([d.strftime("%Y%m%d"), ch, s, int(s * .8), int(s * .03), rng.uniform(.3, .6)])
    return pd.DataFrame(rows, columns=["date", "channel", "sessions", "users", "conversions", "bounce_rate"])

# ---------- transform + validate ----------
def transform(df):
    df = df.copy()
    df["date"] = pd.to_datetime(df["date"], format="%Y%m%d").dt.date.astype(str)
    for c in ["sessions", "users", "conversions"]: df[c] = pd.to_numeric(df[c])
    df["bounce_rate"] = pd.to_numeric(df["bounce_rate"])
    df["conv_rate"] = (df.conversions / df.sessions.replace(0, np.nan)).round(4)
    df["loaded_at"] = dt.datetime.now(dt.timezone.utc).isoformat(timespec="seconds")
    return df

def validate(df):
    issues = []
    if df.empty: issues.append("extract returned no rows")
    if df.duplicated(["date", "channel"]).any(): issues.append("duplicate date/channel rows")
    if (df[["sessions", "users"]] < 0).any().any(): issues.append("negative counts")
    if not df.empty and df.date.max() < str(dt.date.today() - dt.timedelta(days=2)): issues.append("data is stale")
    return issues

# ---------- load (idempotent upsert by date+channel) ----------
def load(df):
    if os.getenv("WAREHOUSE", "sqlite") == "bigquery":
        from google.cloud import bigquery
        cl, table = bigquery.Client(), os.environ["BQ_TABLE"]
        tmp = table + "_staging"
        cl.load_table_from_dataframe(df, tmp, job_config=bigquery.LoadJobConfig(write_disposition="WRITE_TRUNCATE")).result()
        cl.query(f"MERGE `{table}` t USING `{tmp}` s ON t.date=s.date AND t.channel=s.channel "
                 "WHEN MATCHED THEN UPDATE SET sessions=s.sessions, users=s.users, conversions=s.conversions, "
                 "bounce_rate=s.bounce_rate, conv_rate=s.conv_rate, loaded_at=s.loaded_at "
                 "WHEN NOT MATCHED THEN INSERT ROW").result()
        return
    con = sqlite3.connect(os.getenv("SQLITE_PATH", "warehouse.db"))
    con.execute("""CREATE TABLE IF NOT EXISTS web_daily(date TEXT, channel TEXT, sessions INT, users INT,
        conversions INT, bounce_rate REAL, conv_rate REAL, loaded_at TEXT, PRIMARY KEY(date, channel))""")
    con.executemany("INSERT OR REPLACE INTO web_daily VALUES(?,?,?,?,?,?,?,?)",
                    df[["date", "channel", "sessions", "users", "conversions", "bounce_rate", "conv_rate", "loaded_at"]]
                    .itertuples(index=False, name=None))
    con.commit(); con.close()

# ---------- alerts ----------
def anomalies(df):
    out, last = [], df.date.max()
    for ch, g in df.groupby("channel"):
        g = g.sort_values("date"); hist, today = g.iloc[:-1].sessions, g.iloc[-1]
        mean, sd = hist.tail(14).mean(), hist.tail(14).std() or 1
        drop, z = 1 - today.sessions / mean, (today.sessions - mean) / sd
        if drop > DROP_THRESHOLD or abs(z) > Z_LIMIT:
            out.append(f"{ch} sessions on {last}: {int(today.sessions)} vs 14-day avg {mean:.0f} ({-drop:+.0%}, z={z:.1f})")
    return out

def alert(title, lines):
    msg = f":rotating_light: *{title}*\n" + "\n".join(f"• {l}" for l in lines)
    hook = os.getenv("SLACK_WEBHOOK")
    if hook: retry(lambda: requests.post(hook, json={"text": msg}, timeout=10).raise_for_status())
    else: log.error("ALERT (no SLACK_WEBHOOK set)\n%s", msg)

def main():
    try:
        raw = retry(extract_ga4 if os.getenv("SOURCE", "mock") == "ga4" else extract_mock)
        df = transform(raw)
        problems = validate(df)
        if problems: alert("Analytics pipeline: data quality failed", problems); return 1
        retry(lambda: load(df))
        log.info("loaded %d rows", len(df))
        found = anomalies(df)
        if found: alert("Analytics anomaly detected", found)
        return 0
    except Exception as e:
        alert("Analytics pipeline crashed", [repr(e)]); return 2

if __name__ == "__main__":
    raise SystemExit(main())
